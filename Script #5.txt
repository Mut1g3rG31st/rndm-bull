local function getActiveWeaponProfile()
    if _G.WeaponProfile ~= "Auto" then
        for _, p in pairs(_G.WeaponProfiles) do
            if p.name == _G.WeaponProfile then
                return p
            end
        end
    end

    -- Auto-Detect: Prüfe den aktuell gehaltenen Tool-Namen
    local player = game:GetService("Players").LocalPlayer
    local char = player and player.Character
    local tool = char and char:FindFirstChildOfClass("Tool")

    if tool then
        local toolName = tool.Name:lower()
        for _, profile in pairs(_G.WeaponProfiles) do
            if #profile.match > 0 then
                for _, keyword in pairs(profile.match) do
                    if toolName:find(keyword:lower(), 1, true) then
                        return profile
                    end
                end
            end
        end
    end

    -- Fallback: Standard
    return _G.WeaponProfiles[1]
end

local function saveConfig()
    local data = {}
    for _, key in pairs(CONFIG_KEYS) do
        data[key] = _G[key]
    end
    local json = game:GetService("HttpService"):JSONEncode(data)
    -- Speichert in die WriteFile API (Level 2+) oder gibt JSON aus zum Copy
    if writefile then
        writefile("SolaraConfig.json", json)
        return "Gespeichert: SolaraConfig.json"
    elseif makefolder then
        -- Level 2 Fallback
        return json
    else
        return "JSON (Copy):\n" .. json
    end
end

local function loadConfig()
    if readfile then
        local content = readfile("SolaraConfig.json")
        if content then
            local data = game:GetService("HttpService"):JSONDecode(content)
            for key, val in pairs(data) do
                _G[key] = val
            end
            return "Geladen: SolaraConfig.json"
        end
    end
    return "Konfiguration nicht gefunden."
end

-- VISIBILITY HELPER
-- ========================================================
local function isPartVisible(part, character)
    if not part or not character then return false end
    local cam = workspace.CurrentCamera
    if not cam then return false end
    local castPoints = { cam.CFrame.Position, part.Position }
    local ignoreList = { game:GetService("Players").LocalPlayer.Character, character }
    local rays = cam:GetPartsObscuringTarget(castPoints, ignoreList)
    return #rays == 0
end

-- ESP DRAWING SYSTEM
-- ========================================================
local ActiveDrawings = {}

local function destroyElement(obj)
    if obj then
        pcall(function()
            obj.Visible = false
            obj:Remove()
        end)
    end
end

function clearAllDrawingsGlobally()
    for plName, drawings in pairs(ActiveDrawings) do
        if drawings then
            for key, obj in pairs(drawings) do
                if key == "Bones" then
                    for i = 1, 5 do
                        if obj[i] then
                            destroyElement(obj[i])
                            obj[i] = nil
                        end
                    end
                else
                    destroyElement(obj)
                end
            end
            ActiveDrawings[plName] = nil
        end
    end
end

-- ========================================================
-- HUMANOID HELPER
-- ========================================================
local function getHumanoid()
    if player.Character then
        return player.Character:FindFirstChildOfClass("Humanoid")
    end
    return nil
end

local defaultWalkSpeed, defaultJumpPower = 16, 50

local function updateDefaults()
    local hum = getHumanoid()
    if hum then
        if not _G.SpeedEnabled then
            defaultWalkSpeed = hum.WalkSpeed
        end
        if not _G.JumpEnabled then
            defaultJumpPower = hum.JumpPower
        end
    end
end

local flyBV, flyBG

local function toggleFly()
    _G.Flying = not _G.Flying
    FlyButton.Text = _G.Flying and "Fly: AN" or "Fly: AUS"
    FlyButton.BackgroundColor3 = _G.Flying and Color3.fromRGB(45, 180, 90) or Color3.fromRGB(200, 50, 60)

    local char = player.Character
    if _G.Flying and char and char:FindFirstChild("HumanoidRootPart") then
        local hrp = char.HumanoidRootPart
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand = true end

        flyBV = Instance.new("BodyVelocity")
        flyBV.Velocity = Vector3.new(0, 0, 0)
        flyBV.MaxForce = Vector3.new(1, 1, 1) * 10000000
        flyBV.Parent = hrp

        flyBG = Instance.new("BodyGyro")
        flyBG.MaxTorque = Vector3.new(1, 1, 1) * 10000000
        flyBG.P = 9000
        flyBG.CFrame = hrp.CFrame
        flyBG.Parent = hrp

        task.spawn(function()
            while _G.Flying and char and char:FindFirstChild("HumanoidRootPart") and flyBV and flyBG do
                local speed = _G.SpeedEnabled and tonumber(SpeedInput.Text) or 60
                local moveDir = Vector3.new(0, 0, 0)

                if uis:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + camera.CFrame.LookVector end
                if uis:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - camera.CFrame.LookVector end
                if uis:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - camera.CFrame.RightVector end
                if uis:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + camera.CFrame.RightVector end
                if uis:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
                if uis:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir = moveDir - Vector3.new(0, 1, 0) end

                flyBV.Velocity = moveDir * speed
                flyBG.CFrame = camera.CFrame
                task.wait()
            end

            if flyBV then flyBV:Destroy(); flyBV = nil end
            if flyBG then flyBG:Destroy(); flyBG = nil end
            if hum then hum.PlatformStand = false end
        end)
    else
        if flyBV then flyBV:Destroy(); flyBV = nil end
        if flyBG then flyBG:Destroy(); flyBG = nil end
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand = false end
    end
end

FlyButton.MouseButton1Click:Connect(toggleFly)

local function toggleFreecam()
    _G.Freecam = not _G.Freecam
    FreecamButton.Text = "Freecam" .. (_G.Freecam and ": AN" or ": AUS")
    FreecamButton.BackgroundColor3 = _G.Freecam and Color3.fromRGB(45, 180, 90) or Color3.fromRGB(200, 50, 60)

    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")

    if _G.Freecam then
        if hrp then hrp.Anchored = true end
        camera.CameraType = Enum.CameraType.Scriptable

        local cameraRotX = 0
        local cameraRotY = 0
        local _, _, _, R00, R01, R02, _, _, R12, _, _, R22 = camera.CFrame:GetComponents()
        cameraRotX = math.atan2(-R12, R22)
        cameraRotY = math.asin(R02)

        task.spawn(function()
            while _G.Freecam do
                local speed = tonumber(FreecamSpeedInput.Text) or 1.2
                local moveDir = Vector3.new(0, 0, 0)

                if uis:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
                    uis.MouseBehavior = Enum.MouseBehavior.LockCurrentPosition
                    local delta = uis:GetMouseDelta()
                    cameraRotX = cameraRotX - (delta.X * 0.003)
                    cameraRotY = math.clamp(cameraRotY - (delta.Y * 0.003), -math.pi / 2.1, math.pi / 2.1)
                    camera.CFrame = CFrame.new(camera.CFrame.Position)
                        * CFrame.Angles(0, cameraRotX, 0)
                        * CFrame.Angles(cameraRotY, 0, 0)
                else
                    uis.MouseBehavior = Enum.MouseBehavior.Default
                end

                local forwardVector = camera.CFrame.LookVector
                local rightVector = camera.CFrame.RightVector

                if uis:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + forwardVector end
                if uis:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - forwardVector end
                if uis:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - rightVector end
                if uis:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + rightVector end
                if uis:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
                if uis:IsKeyDown(Enum.KeyCode.LeftControl) then moveDir = moveDir - Vector3.new(0, 1, 0) end

                if moveDir.Magnitude > 0 then
                    camera.CFrame = camera.CFrame + (moveDir.Unit * speed)
                end
                task.wait()
            end
            uis.MouseBehavior = Enum.MouseBehavior.Default
        end)
    else
        camera.CameraType = Enum.CameraType.Custom
        uis.MouseBehavior = Enum.MouseBehavior.Default
        if hrp then hrp.Anchored = false end
    end
end

FreecamButton.MouseButton1Click:Connect(toggleFreecam)

-- CUSTOM CROSSHAIR SYSTEM
-- ========================================================
local CrosshairTop, CrosshairBottom, CrosshairLeft, CrosshairRight

local function createCrosshairLines()
    local function parseColor(str)
        local r, g, b = string.match(str, "(%d+)%s*,%s*(%d+)%s*,%s*(%d+)")
        if r and g and b then
            return Color3.fromRGB(tonumber(r), tonumber(g), tonumber(b))
        end
        return Color3.new(1, 1, 1)
    end

    local color = parseColor(_G.CrosshairColor)
    local gap = _G.CrosshairGap
    local length = _G.CrosshairSize
    local thickness = _G.CrosshairThickness

    if not CrosshairTop then
        CrosshairTop = Drawing.new("Line")
        CrosshairTop.Thickness = thickness
        CrosshairBottom = Drawing.new("Line")
        CrosshairBottom.Thickness = thickness
        CrosshairLeft = Drawing.new("Line")
        CrosshairLeft.Thickness = thickness
        CrosshairRight = Drawing.new("Line")
        CrosshairRight.Thickness = thickness
    end

    CrosshairTop.Color = color
    CrosshairBottom.Color = color
    CrosshairLeft.Color = color
    CrosshairRight.Color = color

    local center = camera.ViewportSize / 2
    local inset = guiService:GetGuiInset().Y
    local cy = (camera.ViewportSize.Y - inset) / 2
    local cx = camera.ViewportSize.X / 2

    CrosshairTop.From = Vector2.new(cx, cy - gap)
    CrosshairTop.To = Vector2.new(cx, cy - gap - length)
    CrosshairBottom.From = Vector2.new(cx, cy + gap)
    CrosshairBottom.To = Vector2.new(cx, cy + gap + length)
    CrosshairLeft.From = Vector2.new(cx - gap, cy)
    CrosshairLeft.To = Vector2.new(cx - gap - length, cy)
    CrosshairRight.From = Vector2.new(cx + gap, cy)
    CrosshairRight.To = Vector2.new(cx + gap + length, cy)
end

runService.RenderStepped:Connect(function()
    if _G.CustomCrosshair then
        createCrosshairLines()
        CrosshairTop.Visible = true
        CrosshairBottom.Visible = true
        CrosshairLeft.Visible = true
        CrosshairRight.Visible = true
    else
        if CrosshairTop then CrosshairTop.Visible = false end
        if CrosshairBottom then CrosshairBottom.Visible = false end
        if CrosshairLeft then CrosshairLeft.Visible = false end
        if CrosshairRight then CrosshairRight.Visible = false end
    end
end)

-- NO RECOIL / NO SPREAD / NO FOV KICK / NO SCOPE
-- ========================================================
runService.RenderStepped:Connect(function()
    local char = player.Character
    if not char then return end

    -- No Recoil: Reset Kamera-Position (verhindert Camera-Shake)
    if _G.NoRecoil then
        local tool = char:FindFirstChildOfClass("Tool")
        if tool then
            -- Versuche Recoil-Properties zu nullen
            for _, prop in pairs({ "Recoil", "RecoilAmount", "KickAmount" }) do
                if tool[prop] ~= nil and typeof(tool[prop]) == "number" then
                    tool[prop] = 0
                end
            end
            -- Setze Spread auf 0 falls vorhanden
            if tool.Spread ~= nil and typeof(tool.Spread) == "number" then
                tool.Spread = 0
            end
            if tool.BulletSpread ~= nil and typeof(tool.BulletSpread) == "number" then
                tool.BulletSpread = 0
            end
        end
    end

    -- No FOV Kick: FOV pro Frame zurücksetzen
    if _G.NoFovKick and not _G.FovEnabled then
        camera.FieldOfView = 70
    end

    -- No Scope: Scope-Attachment entfernen
    if _G.NoScope then
        local tool = char:FindFirstChildOfClass("Tool")
        if tool then
            for _, descendant in pairs(tool:GetDescendants()) do
                if descendant:IsA("SurfaceGui") or descendant.Name:find("Scope", 1, true) then
                    if descendant:IsA("BillboardGui") or descendant:IsA("SurfaceGui") then
                        descendant.Enabled = false
                    end
                end
            end
            -- ADS-Delay: Setze ADSProperties
            if tool.AdsEnabled ~= nil then tool.AdsEnabled = false end
            if tool.CanZoom ~= nil then tool.CanZoom = false end
        end
    end

    -- No Falling Damage
    if _G.NoFallingDamage then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            if hum.FallingDamageFactor ~= nil then
                hum.FallingDamageFactor = 0
            end
        end
    end

    -- No Footsteps: Alle Character-Sounds stummschalten
    if _G.NoFootsteps then
        for _, descendant in pairs(char:GetDescendants()) do
            if descendant:IsA("Sound") then
                descendant.Volume = 0
            end
        end
    end
end)

-- ========================================================
-- SMART DODGE (ANTI-AIMBOT SYSTEM)
-- ========================================================
local dodgeLastTime = 0
local dodgeCooldown = 1.5 -- Sekunden zwischen Ausweichbewegungen

runService.Heartbeat:Connect(function()
    if not _G.SmartDodge then return end
    if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then return end
    if _G.Flying or _G.Freecam then return end

    local now = os.clock()
    if now - dodgeLastTime < dodgeCooldown then return end

    local myHrp = player.Character.HumanoidRootPart
    local myPos = myHrp.Position
    local myHum = getHumanoid()
    if not myHum or myHum.Health <= 0 then return end

    local dodgeThreshold = math.rad(tonumber(_G.SmartDodgeSensitivity) or 20)
    local dodgeDirection = Vector3.new(0, 0, 0)
    local found = false

    for _, pl in pairs(players:GetPlayers()) do
        if pl ~= player and pl.Character then
            local theirHrp = pl.Character:FindFirstChild("HumanoidRootPart")
            local theirHum = pl.Character:FindFirstChildOfClass("Humanoid")

            if theirHrp and theirHum and theirHum.Health > 0 then
                -- Nur wenn der Gegner nah genug ist (< 100 studs)
                local distToMe = (myPos - theirHrp.Position).Magnitude
                if distToMe > 100 then continue end

                -- Vektor von Gegner zu mir (horizontal)
                local vectorToMe = myPos - theirHrp.Position
                vectorToMe = Vector3.new(vectorToMe.X, 0, vectorToMe.Z)
                if vectorToMe.Magnitude < 0.1 then continue end
                vectorToMe = vectorToMe.Unit

                -- Richtung, in die der Gegner schaut (horizontal)
                local theirFacing = theirHrp.CFrame.LookVector
                theirFacing = Vector3.new(theirFacing.X, 0, theirFacing.Z)
                if theirFacing.Magnitude < 0.1 then continue end
                theirFacing = theirFacing.Unit

                -- Winkel zwischen ihrer Blickrichtung und dem Vektor zu mir
                local dot = theirFacing:Dot(vectorToMe)
                local angle = math.acos(math.clamp(dot, -1, 1))

                if angle < dodgeThreshold then
                    found = true
                    -- Senkrechte Richtung: ihre Right Vector
                    local rightVec = Vector3.new(theirHrp.CFrame.RightVector.X, 0, theirHrp.CFrame.RightVector.Z)
                    if rightVec.Magnitude > 0.1 then
                        dodgeDirection = dodgeDirection + rightVec.Unit
                    end
                end
            end
        end
    end

    if found and dodgeDirection.Magnitude > 0.1 then
        dodgeLastTime = now
        -- Kräftiger Ausweich-Impuls zur Seite
        local impulse = dodgeDirection.Unit * 15
        myHrp.AssemblyLinearVelocity = myHrp.AssemblyLinearVelocity + impulse
    end
end)

-- SPINBOT WITH JITTER & FAKE LAG COMP

local spinbotLastJitterTime = 0
local spinbotLastLagCompTime = 0
local spinbotLagCompUntil = 0

runService.Heartbeat:Connect(function()
    if not _G.Spinbot then return end
    if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then return end
    if _G.Freecam then return end

    local hrp = player.Character.HumanoidRootPart
    local speed = tonumber(SpinbotSpeedInput.Text) or 50

    local now = os.clock()

    -- Jitter: Zufällige Winkelsprünge alle ~0.3s
    if _G.SpinbotJitter and now - spinbotLastJitterTime > 0.3 then
        spinbotLastJitterTime = now
        local jitterAngle = (math.random() * 2 - 1) * math.rad(15 + speed * 0.2)
        hrp.CFrame = hrp.CFrame * CFrame.Angles(0, jitterAngle, 0)
    end

    -- Fake Lag Comp: Alle ~2s kurz zurückrotieren
    if _G.SpinbotFakeLagComp and now - spinbotLastLagCompTime > 2.0 then
        spinbotLastLagCompTime = now
        spinbotLagCompUntil = now + 0.15
        hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(-speed * 0.8), 0)
    end

    -- Basis-Rotation (pausiert während Lag-Comp-Impuls)
    if now >= spinbotLagCompUntil then
        hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(speed * 0.016), 0)
    end
end)

local function getTargetPart(character, bodyPart)
    if bodyPart == "Torso" then
        return character:FindFirstChild("UpperTorso")
            or character:FindFirstChild("Torso")
            or character:FindFirstChild("HumanoidRootPart")
    elseif bodyPart == "Legs" then
        return character:FindFirstChild("LeftUpperLeg")
            or character:FindFirstChild("RightUpperLeg")
            or character:FindFirstChild("Left Leg")
            or character:FindFirstChild("Right Leg")
            or character:FindFirstChild("HumanoidRootPart")
    else

local function getClosestPlayerToCrosshair()
    local closestPlayer = nil
    local shortestPixelDistance = _G.AimbotFOV
    local screenSize = camera.ViewportSize
    local topInsetValue = guiService:GetGuiInset().Y
    local realCrosshairCenter = Vector2.new(screenSize.X / 2, (screenSize.Y - topInsetValue) / 2)

    FOVCircle.Radius = _G.AimbotFOV
    FOVCircle.Position = realCrosshairCenter
    FOVCircle.Visible = _G.AimbotEnabled

    for _, pl in pairs(players:GetPlayers()) do
        if pl ~= player and (pl.Team ~= player.Team or player.Team == nil) then
            local char = pl.Character
            local targetPart = getTargetPart(char, _G.BodyPartTargeting)
            local hum = char and char:FindFirstChildOfClass("Humanoid")

            if targetPart and hum and hum.Health > 0 then
                local screenPos, onScreen = camera:WorldToViewportPoint(targetPart.Position)
                if onScreen then
                    local pixelDistance = (Vector2.new(screenPos.X, screenPos.Y) - realCrosshairCenter).Magnitude
                    if pixelDistance < shortestPixelDistance then
                        closestPlayer = pl
                        shortestPixelDistance = pixelDistance
                    end
                end
            end
        end
    end

    return closestPlayer
end

-- ADVANCED BALLISTIC AIMBOT (WEAPON VARIABLE PREDICTION)
-- ========================================================
runService.RenderStepped:Connect(function()
    if _G.AimbotEnabled and AimbotKeyDown then
        AimbotTarget = getClosestPlayerToCrosshair()

        if AimbotTarget
            and AimbotTarget.Character
            and player.Character then

            local char = AimbotTarget.Character
            local targetPart = getTargetPart(char, _G.BodyPartTargeting)
            local targetHrp = char:FindFirstChild("HumanoidRootPart")

            if targetPart and targetHrp and targetHrp:IsA("BasePart") then
                -- 1. Ziel-Position
                local targetPosition = targetPart.Position

                -- 2. Distanz
                local distance = (targetPosition - camera.CFrame.Position).Magnitude

                -- 3. Waffen-Profil holen (Auto-Detect oder manuell)
                local profile = getActiveWeaponProfile()
                local bulletVelocity = profile.velocity or _G.BulletVelocity
                local bulletGravity = profile.gravity or _G.BulletGravity

                -- 4. Flugzeit berechnen
                local timeToTarget = distance / bulletVelocity

                -- 5. Bewegungsprediction (Vorziehen)
                local targetVelocity = targetHrp.AssemblyLinearVelocity
                targetPosition = targetPosition + (targetVelocity * timeToTarget)

                -- 6. Schwerkraft-Kompensation (Kugelabfall)
                local gravityOffset = 0.5 * bulletGravity * (timeToTarget ^ 2)
                targetPosition = targetPosition + Vector3.new(0, gravityOffset, 0)

                -- 7. Smooth aus GUI
                local smoothVal = tonumber(AimbotSmoothInput.Text) or profile.smooth or 5
                smoothVal = math.clamp(smoothVal, 1, 20)

                -- 8. Kamera ausrichten
                if not _G.Freecam then
                    camera.CameraType = Enum.CameraType.Custom
                    local targetCFrame = CFrame.new(camera.CFrame.Position, targetPosition)

                    if smoothVal == 1 then
                        camera.CFrame = targetCFrame
                    else
                        camera.CFrame = camera.CFrame:Lerp(targetCFrame, 1 / smoothVal)
                    end
                end
            end
        else
            AimbotTarget = nil
            FOVCircle.Visible = false
        end
    else
        if not _G.AimbotEnabled then
            FOVCircle.Visible = false
        end
    end
end)

local function translateText(text, targetLang)
    if not text or text == "" then return text end

    local url = "googleapis.com" .. targetLang .. "&dt=t&q=" .. httpService:UrlEncode(text)

    local success, result = pcall(function()
        return game:HttpGet(url)
    end)

    if success and result then
        if string.sub(result, 1, 1) ~= "[" and string.sub(result, 1, 1) ~= "{" then
            return text
        end

        local decodeSuccess, decoded = pcall(function()
            return httpService:JSONDecode(result)
        end)

        if decodeSuccess and decoded and typeof(decoded) == "table" then
            if decoded[1] and typeof(decoded[1]) == "table"
                and decoded[1][1] and typeof(decoded[1][1]) == "table"
                and decoded[1][1][1] then
                return tostring(decoded[1][1][1])
            end
        end
    end

    return text
end

                        })
                    end
                end)
            end
        end)
    end
end

-- ENTITY KILLER (Name eingeben → TP dorthin oder Destroy)
-- ========================================================
task.spawn(function()
    -- Entity Killer: Geht zu einer Entity oder zerstört sie
    -- Nutzung: _G.EntityTarget = "Name" setzen, dann triggert
    while task.wait(0.5) do
        local target = _G.EntityTarget
        if not target or target == "" then continue end

        local entity = workspace:FindFirstChild(target)
        if not entity then
            -- Auch in Modellen suchen
            for _, model in pairs(workspace:GetChildren()) do
                if model:IsA("Model") then
                    entity = model:FindFirstChild(target)
                    if entity then break end
                end
            end
        end

        if entity then
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                local entPos = entity.Position or (entity:IsA("BasePart") and entity.Position) or nil
                if entPos then
                    player.Character.HumanoidRootPart.CFrame = CFrame.new(entPos + Vector3.new(0, 5, 0))
                end
            end

            -- Optional: Destroy
            if _G.EntityKill then
                pcall(function() entity:Destroy() end)
            end

            _G.EntityTarget = "" -- Reset nach Nutzung
        end
    end
end)

-- ========================================================
-- AUTO-DRIVE (Game-Specific: Car Dealership Tycoon etc.)
-- ========================================================
_G.AutoDrive = false
_G.AutoDriveSpeed = 60

task.spawn(function()
    while task.wait(0.1) do
        if not _G.AutoDrive then continue end
        if not player.Character then continue end

        -- Sucht ein Fahrzeug (VehicleSeat oder mit "Car"/"Vehicle" im Namen)
        local seat = nil
        local car = nil

        for _, model in pairs(workspace:GetChildren()) do
            if model:IsA("Model") then
                seat = model:FindFirstChildWhichIsA("VehicleSeat")
                if seat then
                    car = model
                    break
                end
            end
        end

        -- Fallback: Nächstes VehicleSeat in der Nähe
        if not seat then
            local closest = nil
            local closestDist = 200
            for _, obj in pairs(workspace:GetDescendants()) do
                if obj:IsA("VehicleSeat") then
                    local d = (obj.Position - (player.Character.HumanoidRootPart or obj).Position).Magnitude
                    if d < closestDist then
                        closestDist = d
                        closest = obj
                        car = closest.Parent
                    end
                end
            end
            if closest then seat = closest end
        end

        if seat and car then
            local speed = tonumber(_G.AutoDriveSpeed) or 60

            -- Wacht auf WASD-Eingaben über die Seat-Properties
            if uis:IsKeyDown(Enum.KeyCode.W) then
                seat.Throttle = 1
            elseif uis:IsKeyDown(Enum.KeyCode.S) then
                seat.Throttle = -1
            else
                seat.Throttle = 0
            end

            if uis:IsKeyDown(Enum.KeyCode.A) then
                seat.Steering = 1
            elseif uis:IsKeyDown(Enum.KeyCode.D) then
                seat.Steering = -1
            else
                seat.Steering = 0
            end

            -- Speed-Limit durchsetzen
            local currentSpeed = car.AssemblyLinearVelocity.Magnitude
            if currentSpeed > speed then
                seat.Throttle = 0
            end
        end
    end
end)
